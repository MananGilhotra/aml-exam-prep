→ Lecture 1 theory

From scratch

def min_max_scale(x):
    """(X - X_min) / (X_max - X_min)  ->  maps to [0, 1]"""
    x = np.asarray(x, dtype=float)
    return (x - x.min()) / (x.max() - x.min())


def z_score_scale(x):
    """(X - mu) / sigma  ->  mean 0, std 1"""
    x = np.asarray(x, dtype=float)
    return (x - x.mean()) / x.std()


def max_abs_scale(x):
    """X / max(|X|)  ->  maps to [-1, 1], PRESERVES zeros (good for sparse)"""
    x = np.asarray(x, dtype=float)
    return x / np.max(np.abs(x))


def robust_scale(x):
    """(X - median) / IQR  ->  unaffected by outliers"""
    x  = np.asarray(x, dtype=float)
    q1 = np.percentile(x, 25)
    q3 = np.percentile(x, 75)
    return (x - np.median(x)) / (q3 - q1)


# Demonstrating why Min-Max fails with an outlier
x = np.array([100, 150, 200, 250, 50000])
print(min_max_scale(x))   # [0. 0.001 0.002 0.003 1.] -> normals collapse
print(robust_scale(x))    # [-0.004 -0.002 0. 0.002 1.992] -> normals stay distinct

L3Simple Linear Regression (OLS)

→ Lecture 3 theory  ·  from Lab_3/slr_ols_todo.py

def find_mean(values):
    return np.sum(values) / len(values)


def find_slope_intercept(x, y, x_mean, y_mean):
    """OLS closed form:  m = sum((x-xbar)(y-ybar)) / sum((x-xbar)^2),  c = ybar - m*xbar"""
    numerator   = 0
    denominator = 0
    for x1, y1 in zip(x, y):
        numerator   += (x1 - x_mean) * (y1 - y_mean)
        denominator += (x1 - x_mean) ** 2
    m = numerator / denominator
    b = y_mean - m * x_mean
    return m, b


def predict_lr(x, m, b):
    return m * x + b


def calculate_residuals(y, y_pred):
    return y - y_pred


# --- Evaluation metrics (also used in Labs 4, 7, 8) ---
def mae(y, y_pred):
    return np.mean(np.abs(y - y_pred))

def mse(y, y_pred):
    return np.mean((y - y_pred) ** 2)

def rmse(y, y_pred):
    return np.sqrt(mse(y, y_pred))

def r_square(y, y_pred, y_mean):
    ss_total    = np.sum((y - y_mean) ** 2)     # TSS
    ss_residual = np.sum((y - y_pred) ** 2)     # RSS
    return 1 - (ss_residual / ss_total)

L4Multiple Linear Regression (Matrix OLS)

→ Lecture 4 theory  ·  from Lab_4/mlr_ols_student.py

def fit_multiple_lr(X, y):
    """beta* = (X^T X)^-1 X^T y   -- the Normal Equation, in one line"""
    ones = np.ones((len(X), 1))
    X    = np.hstack((ones, X))               # prepend the intercept column
    beta = np.linalg.inv(X.T @ X) @ X.T @ y
    return beta


def fit_multiple_lr_beta(X, y):
    """Same thing, written step by step so each matrix is visible."""
    ones = np.ones((len(X), 1))
    X    = np.hstack((ones, X))
    XT           = X.T                        # (m+1) x n
    XTX          = XT @ X                     # (m+1) x (m+1)  <- SQUARE, symmetric
    XTX_inverse  = np.linalg.inv(XTX)         # fails if det == 0 (multicollinearity)
    XTy          = XT @ y                     # (m+1) x 1
    beta         = XTX_inverse @ XTy
    return beta


def predict_multiple_lr(X, beta):
    ones = np.ones((len(X), 1))
    X    = np.hstack((ones, X))
    return X @ beta
What happens with perfect multicollinearity
X = np.array([[1., 1., 2.],
              [1., 2., 3.],
              [1., 3., 4.]])        # col3 = col1 + col2
print(np.linalg.det(X.T @ X))       # 0.0  (up to float noise)
np.linalg.inv(X.T @ X)              # LinAlgError: Singular matrix

Safer in practice: use np.linalg.pinv (pseudo-inverse) or np.linalg.lstsq, which return a minimum-norm solution instead of crashing. Note that this hides the problem — the coefficients are still not uniquely identified.

L5–L6Gradient Descent (Batch, SGD, Mini-Batch)

→ Lecture 5 · Lecture 6  ·  from Lab_6/mlr_gradient_descent_minibatch.py

The four building blocks

def predict_mlr(X, weights):
    """y_hat = H @ weights,  where H = [1 | X]"""
    ones = np.ones((len(X), 1))
    H = np.hstack((ones, X))
    return H @ weights


def compute_mlr_residual(y, predictions):
    """predicted MINUS actual  (note the sign convention used in the gradient)"""
    return predictions - y


def compute_mlr_gradient(X, residual):
    """grad L = (2/m) * H^T (H@theta - y)"""
    ones = np.ones((len(X), 1))
    H = np.hstack((ones, X))
    return 2 / len(X) * (H.T @ residual)


def compute_mlr_loss(residual):
    return np.mean(residual ** 2)              # MSE


def update_mlr_parameters(weights, gradient, alpha):
    return weights - alpha * gradient          # MINUS: gradient points uphill

Batch Gradient Descent — 1 update per epoch

def batch_gradient_descent(X, y, theta, alpha, epochs):
    loss_history = []
    for epoch in range(epochs):
        pred     = predict_mlr(X, theta)              # uses the FULL dataset
        residual = compute_mlr_residual(y, pred)
        grad     = compute_mlr_gradient(X, residual)
        theta    = update_mlr_parameters(theta, grad, alpha)   # ONE update
        loss_history.append(compute_mlr_loss(residual))
    return theta, loss_history

Mini-Batch Gradient Descent — ceil(m/b) updates per epoch

def minibatch_gradient_descent_mlr(X, y, theta, alpha, epochs, batch_size):
    loss_history = []
    n = X.shape[0]

    for epoch in range(epochs):
        indices    = np.random.permutation(n)        # SHUFFLE every epoch
        X_shuffled = X[indices]
        y_shuffled = y[indices]

        for start in range(0, n, batch_size):        # ceil(n/b) mini-batches
            end     = start + batch_size
            X_batch = X_shuffled[start:end]          # last batch may be smaller
            y_batch = y_shuffled[start:end]

            pred     = predict_mlr(X_batch, theta)
            residual = compute_mlr_residual(y_batch, pred)
            grad     = compute_mlr_gradient(X_batch, residual)
            theta    = update_mlr_parameters(theta, grad, alpha)   # update INSIDE the loop

        # track the FULL-dataset loss once per epoch, for the curve
        residual = compute_mlr_residual(y, predict_mlr(X, theta))
        loss_history.append(compute_mlr_loss(residual))

    return theta, loss_history


# SGD is exactly this with batch_size = 1
theta_sgd, _ = minibatch_gradient_descent_mlr(X, y, theta0, alpha=0.01,
                                              epochs=50, batch_size=1)
The single line that separates Batch GD from SGD

The theta = update_mlr_parameters(...) line. In Batch GD it sits outside the sample loop (once per epoch). In SGD / Mini-Batch it sits inside. Accumulating gradients through the whole inner loop before updating silently turns your SGD back into Batch GD.

Regression — from scratch

def mape(y, y_pred):
    """UNDEFINED where y == 0 -- guard it explicitly"""
    y, y_pred = np.asarray(y, float), np.asarray(y_pred, float)
    if np.any(y == 0):
        raise ValueError("MAPE is undefined when any actual value is zero")
    return 100 * np.mean(np.abs((y - y_pred) / y))


def adjusted_r_square(y, y_pred, n, p):
    """p = number of predictors, EXCLUDING the intercept. Requires n > p + 1."""
    rss = np.sum((y - y_pred) ** 2)
    tss = np.sum((y - np.mean(y)) ** 2)
    # use rss/tss directly instead of a rounded r2 -- more accurate
    return 1 - (rss / tss) * ((n - 1) / (n - p - 1))

Classification — confusion matrix by hand and with sklearn

def binary_counts(y_true, y_pred, positive_class):
    """Collapse a multiclass problem to 'positive_class vs rest'."""
    tp = np.sum((y_true == positive_class) & (y_pred == positive_class))
    fn = np.sum((y_true == positive_class) & (y_pred != positive_class))
    fp = np.sum((y_true != positive_class) & (y_pred == positive_class))
    tn = np.sum((y_true != positive_class) & (y_pred != positive_class))
    return tp, fp, fn, tn


def precision_recall_f1(tp, fp, fn):
    precision = tp / (tp + fp) if (tp + fp) else float('nan')   # PREDICTED positives
    recall    = tp / (tp + fn) if (tp + fn) else float('nan')   # ACTUAL positives
    f1        = 2 * tp / (2 * tp + fp + fn)                     # never uses TN
    return precision, recall, f1


# sklearn: rows = actual, columns = predicted (same convention as the worksheet)

Reading classification_report in the viva

"accuracy" row = micro average (in single-label multiclass they are equal). "macro avg" = every class equal weight. "weighted avg" = weighted by support (actual class size). If your minority class matters operationally, quote macro, not weighted.

L9Bias–Variance, CV & learning curves

→ Lecture 9 theory  ·  from Lab_9/bias_variance_tools.py + notebook

The controlled simulation (true $f$ is known)

def true_function(x):
    """The hidden pattern we are trying to learn: f(x) = x^2."""
    return np.asarray(x, dtype=float) ** 2


def collect_repeated_predictions(degree, x_values, n_models, sample_size,
                                 noise_sd, seed):
    """Simulate many training datasets, fit one model to each, record predictions.
    Repeating this is how we approximate E[f_hat(x)] and Var(f_hat(x))."""
    rng   = np.random.default_rng(seed)
    preds = []
    for _ in range(n_models):
        x_s = rng.uniform(-15, 10, sample_size)
        y_s = true_function(x_s) + rng.normal(0, noise_sd, sample_size)
        coef = np.polyfit(x_s, y_s, degree)
        preds.append(np.polyval(coef, x_values))
    return np.array(preds)                  # shape (n_models, len(x_values))


# --- The decomposition: MSE = Bias^2 + Variance + Var(noise) ---
preds      = collect_repeated_predictions(degree=2, x_values=x_grid,
                                          n_models=100, sample_size=30,
                                          noise_sd=20, seed=42)
mean_pred  = preds.mean(axis=0)                       # E[f_hat(x)]
bias_sq    = np.mean((mean_pred - true_function(x_grid)) ** 2)
variance   = np.mean(preds.var(axis=0))
noise      = 20 ** 2                                  # Var(epsilon), known here
print(f"Bias^2={bias_sq:.1f}  Var={variance:.1f}  Noise={noise}  MSE={bias_sq+variance+noise:.1f}")

From scratch — every step of the worksheet

def pca_from_scratch(X, k):
    X = np.asarray(X, dtype=float)
    n, p = X.shape

    # Step 1: mean centering  ->  X_c = X - 1 mu^T   (COMPULSORY)
    mu   = X.mean(axis=0)                       # shape (p,)
    X_c  = X - mu                               # numpy broadcasts the outer product

    # Step 2: covariance matrix  ->  S = (1/(n-1)) X_c^T X_c    shape (p, p)
    S = (X_c.T @ X_c) / (n - 1)
    # equivalently: S = np.cov(X_c, rowvar=False)

    # Step 3: eigendecomposition. eigh() is for SYMMETRIC matrices -> real eigenvalues
    eigvals, eigvecs = np.linalg.eigh(S)

    # Step 4: sort by DESCENDING eigenvalue (eigh returns ascending)
    order   = np.argsort(eigvals)[::-1]
    eigvals = eigvals[order]
    eigvecs = eigvecs[:, order]

    # Step 5: keep the top k, project
    W_k = eigvecs[:, :k]                        # shape (p, k)
    Z   = X_c @ W_k                             # shape (n, k)

    evr     = eigvals / eigvals.sum()           # explained variance ratio
    cum_evr = np.cumsum(evr)
    return Z, W_k, mu, eigvals, evr, cum_evr

🎤 Likely viva questions (and one-line answers)

QuestionAnswer
Why do you prepend a column of ones to $\mathbf X$?So that the intercept $\beta_0$ appears in every row: $1 \times \beta_0 = \beta_0$.
Why does np.linalg.inv crash on your data?$\det(\mathbf X^\top\mathbf X) = 0$ — the columns are linearly dependent (multicollinearity).
Why use np.linalg.lstsq instead of inv?It is numerically more stable and returns a solution even for a singular system.
Where exactly does Batch GD differ from SGD in code?The position of the $\boldsymbol\theta$-update line: outside the sample loop vs inside it.
Why np.random.permutation every epoch?Shuffling prevents biased update sequences when the dataset is ordered.
Why eigh and not eig for PCA?eigh is for symmetric matrices; $S$ is symmetric, so it returns real eigenvalues and orthogonal eigenvectors, and is faster.
Why [::-1] after argsort in PCA?eigh returns eigenvalues in ascending order; PCA needs them descending.
Why is Pipeline used for cross-validation?It refits every preprocessing step inside each fold, which prevents data leakage.
Why scale before SGDRegressor but not before LinearRegression?GD's gradient scales with feature magnitude, so one large feature dominates. The closed-form OLS solution is scale-invariant.
Why does PolynomialFeatures(degree=10) need scaling?$x^{10}$ produces enormous values that cause numerical overflow and an ill-conditioned $\mathbf X^\top\mathbf X$.
